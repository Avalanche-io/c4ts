# Windows: making a symlink needs a privilege the runner's user lacks.
win32 packages/core/tests/scan-symlinks.test.ts > scan records symbolic links > a link to its own directory does not recurse  # no symlinks without the privilege
win32 packages/core/tests/scan-symlinks.test.ts > scan records symbolic links > records each kind of link with Go's target and ID rules  # no symlinks without the privilege

# Windows: chmod 000 leaves a file readable, so nothing is unreadable.
win32 packages/core/tests/scan-unreadable.test.ts > scan with an unreadable file > records it with a null C4 ID and reports it  # chmod cannot make a file unreadable

# Windows: no POSIX modes, so the sub-second vector's scan differs from
# saved.c4m by mode alone (the vector's README, rule 2); its times and its
# diff against its own saved c4m are still checked there.
win32 packages/node/tests/subsecond-vector.test.ts > sub-second vector > the scan is saved.c4m byte for byte, a diff against it is empty, and it resolves to root-id.txt  # no POSIX modes on Windows
win32 packages/node/tests/subsecond-vector.test.ts > sub-second vector > before 1970 the scan floors toward negative infinity  # Node on Windows reads a pre-1970 mtime 2^32 s late (measured: -1.5 s reads as 4294967294.5 s)

# Windows: no POSIX modes, and Node reads the epoch vector's pre-1970 row 2^32 s late.
win32 packages/node/tests/epoch-second-vector.test.ts > epoch-second vector > the scan is saved.c4m byte for byte and resolves to root-id.txt  # no POSIX modes; Node reads pre-1970 times 2^32 s late

# Windows: the 1.0.27 notes command for c4ts-only objects is a POSIX shell line.
